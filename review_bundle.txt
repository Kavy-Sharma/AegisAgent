"""
web/server.py - AegisAgent Web Application Server

Hand-built Flask web server bound strictly to 127.0.0.1:8501.
Replaces the Streamlit frontend with a high-performance, air-gapped web interface.

ROOT_DIR = Path(__file__).resolve().parent.parent


def has_registered_users() -> bool:
    """Checks if any user account has been registered in data/users.json."""
    data = security.auth._load_data()
    return bool(data.get("users"))


def create_app() -> Flask:
    """Flask application factory."""
    app = Flask(
        __name__,
        template_folder=str(ROOT_DIR / "web" / "templates"),
        static_folder=str(ROOT_DIR / "web" / "static"),
    )

    # Cryptographic Session Configuration
    app.secret_key = get_web_secret()
    app.config.update(
        SESSION_COOKIE_HTTPONLY=True,
        SESSION_COOKIE_SAMESITE="Strict",
        SESSION_COOKIE_SECURE=False,  # False for local HTTP 127.0.0.1 execution
        PERMANENT_SESSION_LIFETIME=timedelta(minutes=15),
    )

    # Security Request Lifecycle Hooks
    @app.before_request
    def check_host():
        return verify_host_header()

    @app.before_request
    def check_timeout_and_csrf():
        timeout_res = enforce_session_timeout()
        if timeout_res:
            return timeout_res
        return verify_csrf()

    @app.after_request
    def add_security_headers(response):
        return apply_security_headers(response)

    # Register API Blueprint
    app.register_blueprint(api_bp)

    # Page Routes
    @app.route("/", methods=["GET"])
    def index():
        if session.get("username"):
            return redirect(url_for("overview"))
        if not has_registered_users():
            return redirect(url_for("setup"))
        return redirect(url_for("login"))

    @app.route("/login", methods=["GET", "POST"])
    def login():
        # First-run setup redirect if no users exist
        if not has_registered_users():
            return redirect(url_for("setup"))

        if request.method == "POST":
            username = (request.form.get("username") or "").strip()
            password = request.form.get("password") or ""

            success, res = security.auth.verify_login(username, password)
            if success:
                session.clear()
                session["username"] = username
                session["role"] = res
                session["last_activity"] = time.time()
                session["csrf_token"] = secrets.token_hex(32)
                session["console_session_id"] = "sess_" + secrets.token_hex(8)
                return redirect(url_for("overview"))
            else:
                return render_template("login.html", error=res, expired=False), 401

        # GET request
        if session.get("username"):
            return redirect(url_for("overview"))

        expired = request.args.get("expired") == "1"
        return render_template("login.html", error=None, expired=expired)

    @app.route("/setup", methods=["GET", "POST"])
    def setup():
        # Once any user exists, setup is permanently disabled
        if has_registered_users():
            return redirect(url_for("login"))

        if request.method == "POST":
            username = (request.form.get("username") or "").strip()
            password = request.form.get("password") or ""
            confirm = request.form.get("confirm_password") or ""

            if not username or not password:
                return render_template("setup.html", error="Username and password are required."), 400

            if password != confirm:
                return render_template("setup.html", error="Passwords do not match."), 400

            try:
                security.auth.create_user(username, password, role="admin")
                session.clear()
                session["username"] = username
                session["role"] = "admin"
                session["last_activity"] = time.time()
                session["csrf_token"] = secrets.token_hex(32)
                session["console_session_id"] = "sess_" + secrets.token_hex(8)
                return redirect(url_for("overview"))
            except Exception as e:
                return render_template("setup.html", error=f"Setup failed: {e}"), 400

        return render_template("setup.html", error=None)

    @app.route("/logout", methods=["GET", "POST"])
    def logout():
        username = session.get("username")
        if username:
            security.auth.logout(username)
        session.clear()
        session["csrf_token"] = secrets.token_hex(32)
        return redirect(url_for("login"))

    @app.route("/overview", methods=["GET"])
    @login_required
    def overview():
        return render_template("overview.html", active_page="overview")

    @app.route("/console", methods=["GET"])
    @login_required
    def console():
        health = security.health.get_system_health()
        worker_alive = bool(health.get("worker", {}).get("alive"))
        return render_template("console.html", active_page="console", worker_alive=worker_alive)

    @app.route("/approvals", methods=["GET"])
    @login_required
    @role_required("admin")
    def approvals():
        return render_template("approvals.html", active_page="approvals")

    # Error Handlers
    @app.errorhandler(400)
    def handle_bad_request(e):
        if request.path.startswith("/api/"):
            return jsonify({"error": "Bad Request", "detail": str(e)}), 400
        return f"<h3>400 Bad Request</h3><p>{e}</p>", 400

    @app.errorhandler(403)
    def handle_forbidden(e):
        if request.path.startswith("/api/"):
            return jsonify({"error": "Forbidden", "detail": "Access denied"}), 403
        return "<h3>403 Access Denied</h3><p>Administrative privileges are required to access this resource.</p>", 403

    @app.errorhandler(404)
    def handle_not_found(e):
        if request.path.startswith("/api/"):
            return jsonify({"error": "Endpoint Not Found"}), 404
        return "<h3>404 Page Not Found</h3><p>The requested page does not exist.</p>", 404

    @app.errorhandler(500)
    def handle_internal_error(e):
        if request.path.startswith("/api/"):
            return jsonify({"error": "Internal Server Error"}), 500
        return "<h3>500 Internal Server Error</h3><p>An unexpected error occurred.</p>", 500

    return app


if __name__ == "__main__":

    print("=" * 60)

  function pollTurn(reqId, turnUi) {
    if (currentPollTimer) clearInterval(currentPollTimer);

    currentPollTimer = setInterval(function () {
      api.get("/api/console/poll/" + encodeURIComponent(reqId))
        .then(function (data) {
          var steps = data.steps || [];
          var final = data.final;

          // Update stages in stepper
          steps.forEach(function (step) {
            var node = turnUi.nodeMap[step.stage];
            if (node) {
              node.className = "step-node step-" + (step.status || "pending");
              if (step.detail && Object.keys(step.detail).length > 0) {
                var detailStr = JSON.stringify(step.detail);
                node.title = detailStr;
              }
            }
            var conn = turnUi.connectorMap[step.stage];
            if (conn && step.status === "pass") {
              conn.classList.add("connector-filled");
            }

            // Provide real-time friendly feedback during LLM reasoning
            if (step.stage === "llm" && step.status === "running" && !final && !turnUi.approvalShown) {
              turnUi.textEl.textContent = "AegisAgent reasoning with Ollama (llama3.1)...";
            }
          });

          // Check if approval is actually pending right now
          var lastApproval = null;
          for (var i = steps.length - 1; i >= 0; i--) {
            if (steps[i].stage === "approval") {
              lastApproval = steps[i];
              break;
            }
          }
          var isApprovalPending = lastApproval && lastApproval.status === "running" && !final;
          if (isApprovalPending && !turnUi.approvalShown) {
            var targetTool = "high_risk_tool";
            for (var j = 0; j < steps.length; j++) {
              if (steps[j].detail && steps[j].detail.tool) {
                targetTool = steps[j].detail.tool;
              }
            }
            showInlineApproval(turnUi, reqId, targetTool);
          } else if (!isApprovalPending && turnUi.approvalCard) {
            if (turnUi.approvalCard.parentNode) {
              turnUi.approvalCard.parentNode.removeChild(turnUi.approvalCard);
            }
            turnUi.approvalCard = null;
            turnUi.approvalShown = false;
          }

          if (final) {
            clearInterval(currentPollTimer);
            currentPollTimer = null;
            if (turnUi.approvalCard && turnUi.approvalCard.parentNode) {
              turnUi.approvalCard.parentNode.removeChild(turnUi.approvalCard);
              turnUi.approvalCard = null;
            }

            if (final.decision === "BLOCK") {
              // Clear duplicate bubble text so the callout is the single source of truth
              turnUi.textEl.textContent = "";
              turnUi.callout.className = "callout-blocked";
              turnUi.callout.style.setProperty("display", "block");
              var errDetail = final.error || "Policy violation";
              var isCumulative = errDetail.indexOf("Cumulative suspicion") !== -1;

              var calloutHtml = `
                <div class="flex items-start gap-sm">
                  <div class="flex-1">
                    <div class="font-semibold text-danger">Blocked by AegisAgent: ${errDetail}</div>
                    ${isCumulative ? '<p class="text-xs text-muted mt-xs">Multi-turn protection active: Suspicion was carried over from previous attack attempt to protect IoT hardware.</p>' : ''}
                    ${userRole === "admin" ? `
                      <div class="mt-sm flex items-center gap-xs">
                        <button type="button" class="btn btn-secondary btn-xs btn-clear-threat" id="btn-clear-threat-${reqId}">
                          <svg class="icon icon-xs"><use href="/static/icons.svg#icon-refresh"></use></svg>
                          <span>Clear Threat Window (Acknowledge & Continue)</span>
                        </button>
                      </div>
                    ` : ''}
                  </div>
                </div>
              `;
              turnUi.callout.innerHTML = calloutHtml;

              var clearBtn = document.getElementById("btn-clear-threat-" + reqId);

  function sendPrompt(text) {
    text = (text || "").trim();
    if (!text) return;

    appendUserMessage(text);
    chatTextarea.value = "";
    chatTextarea.style.height = "auto";

    var turnUi = createAssistantTurn();

    api.post("/api/console/send", { text: text })
      .then(function (res) {
        pollTurn(res.id, turnUi);
      })
      .catch(function (err) {
        turnUi.textEl.textContent = "Failed to dispatch request: " + err.message;
      });
  }

  window.sendConsolePrompt = sendPrompt;

  // Event Listeners
  if (chatSendBtn) {
    chatSendBtn.addEventListener("click", function () {
      sendPrompt(chatTextarea.value);
    });
  }

  chatTextarea.addEventListener("keydown", function (e) {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      sendPrompt(chatTextarea.value);
    }
  });

  // Suggested prompt chip clicks
  var chips = document.querySelectorAll(".chip-btn");
  chips.forEach(function (chip) {
    chip.addEventListener("click", function () {
      var prompt = chip.dataset.prompt || chip.textContent.trim();
      sendPrompt(prompt);
    });
  });

  // Start Worker Button on offline banner
  var startWorkerBtn = document.getElementById("btn-start-worker");
  if (startWorkerBtn) {
    startWorkerBtn.addEventListener("click", function () {
      startWorkerBtn.disabled = true;
      startWorkerBtn.classList.add("is-loading");
      api.post("/api/worker/start", {})
        .then(function () {
          app.showToast("Worker Starting", "Worker process initiated.", "ok");
          setTimeout(function () { window.location.reload(); }, 1200);
        })
        .catch(function (err) {
          app.showToast("Worker Error", err.message, "danger");
          startWorkerBtn.disabled = false;
          startWorkerBtn.classList.remove("is-loading");
        });
    });
  }

  // Reset Session Button
  var newSessionBtn = document.getElementById("btn-new-session");
  if (newSessionBtn) {
    newSessionBtn.addEventListener("click", function () {
      newSessionBtn.disabled = true;
      api.post("/api/console/new_session", {})
        .then(function () {
          app.showToast("Session Reset", "Started fresh chat session.", "ok");
          setTimeout(function () { window.location.reload(); }, 500);
        })
        .catch(function (err) {
          app.showToast("Reset Error", err.message, "danger");
          newSessionBtn.disabled = false;
        });
    });
  }

  // Check URL query param for instant prompt
  var urlParams = new URLSearchParams(window.location.search);
  var initialPrompt = urlParams.get("prompt");
  if (initialPrompt) {
    history.replaceState(null, "", window.location.pathname);
    setTimeout(function () {
      sendPrompt(initialPrompt);
    }, 200);
  }
})();

        const fraction = Math.max(0, Math.min(1, diffSec / totalDuration));

        ringFill.style.strokeDashoffset = offset.toFixed(2);
        if (diffSec <= 5) {
          ringFill.style.stroke = "var(--danger)";
        } else if (diffSec <= 12) {
          ringFill.style.stroke = "var(--warn)";
        } else {

      if (diffSec <= 0) {

        buttons.forEach((b) => {
          b.disabled = true;
          b.classList.add("btn-disabled");
        });

/**
 * AegisAgent API Client
 * Wraps fetch with CSRF protection, error handling, and session expiration checks.
 */
var api = (function () {
  function getCsrfToken() {
    var meta = document.querySelector('meta[name="csrf-token"]');
    return meta ? meta.getAttribute("content") : "";
  }

  function request(url, options) {
    options = options || {};
    var headers = options.headers || {};
    var method = (options.method || "GET").toUpperCase();

    headers["Accept"] = "application/json";

    if (method !== "GET" && method !== "HEAD") {
      headers["X-CSRF-Token"] = getCsrfToken();
      if (options.body && typeof options.body === "object" && !(options.body instanceof FormData)) {
        headers["Content-Type"] = "application/json";
        options.body = JSON.stringify(options.body);
      }
    }

    options.headers = headers;
    options.credentials = "same-origin";

    return fetch(url, options)
      .then(function (response) {
        if (response.status === 401) {
          window.location.href = "/login?expired=1";
          return Promise.reject(new Error("Session expired"));
        }
        return response.json().then(function (data) {
          if (!response.ok) {
            var errMsg = (data && data.error) || "Request failed (" + response.status + ")";
            return Promise.reject(new Error(errMsg));
          }
          return data;
        });
      });
  }

  return {
    get: function (url) {
      return request(url, { method: "GET" });
    },
    post: function (url, body) {
      return request(url, { method: "POST", body: body });
    },
    getCsrfToken: getCsrfToken,
    // Helper shortcuts
    getHealth: function () {
      return request("/api/health", { method: "GET" });
    },
    getOverview: function () {
      return request("/api/overview", { method: "GET" });
    },
    getPendingApprovals: function () {
      return request("/api/approvals/pending", { method: "GET" });
    },
    getApprovalHistory: function () {
      return request("/api/approvals/history", { method: "GET" });
    },
    decideApproval: function (requestId, requestHash, approved) {
      return request("/api/approvals/decide", {
        method: "POST",
        body: { request_id: requestId, request_hash: requestHash, approved: approved },
      });
    },
    sendChat: function (text) {
      return request("/api/console/send", {
        method: "POST",
        body: { text: text },
      });
    },
    newSession: function () {
      return request("/api/console/new_session", { method: "POST" });
    },
    clearThreat: function () {

            renderPaletteResults(input.value);

        } else if (e.key === "ArrowUp") {

          if (items.length > 0) {
            selectedIndex = (selectedIndex - 1 + items.length) % items.length;
            renderPaletteResults(input.value);

          e.preventDefault();
          var q = (input.value || "").toLowerCase().trim();

            return !q || cmd.title.toLowerCase().indexOf(q) !== -1;

          closePalette();

"""
Phase 8A & Phase 8A-fix ƒ?" Agent Worker Service

WHAT THIS FILE DOES:
1. Polls data/chat/inbox/*.json every 0.3s for incoming user requests.
2. Validates request IDs and inbox file stems against ^[A-Za-z0-9_-]{8,64}$ before
   building ANY path. Prevents path traversal (e.g. '../../evil').
3. Verifies cryptographic signature using security.signing.
4. Enforces replay protection: rejects if |now - ts| > 60s or nonce was seen before
   (maintains seen_nonces capped at 2000 in memory + data/chat/seen_nonces.json).
5. Enforces valid kind ('chat', 'direct_tool') and roles defined in policy.yaml.
6. Dispatches chat requests to run_turn(text, role, session_id, emit) with role taken
   strictly from the signed request.
7. Dispatches 'direct_tool' requests through gateway.evaluate(..., emit=emit) without
   duplicating policy checks; rate limit counted exactly once.
8. Streams execution steps to data/chat/outbox/<id>.jsonl and writes final reply and decision.
9. Runs worker heartbeat on a daemon thread every 2s with busy and current_request_id fields,
   and refreshes ollama_ok every 15s with a cheap ollama.list() (2s timeout).
10. Maintains MQTT device status and events, recording broker_connected in device_status.json.
11. Handles graceful shutdown on Ctrl+C and never crashes on a single bad request.

class DeviceStatusSubscriber:
    """Maintains a persistent MQTT subscription for hardware status and events."""

    def __init__(self, data_dir: Path):
        self.data_dir = data_dir
        self.client = None
        self.is_running = False
        self.broker_connected = False
        self._last_state = None
        self._since = None

        hw_cfg = security.policy_loader.POLICY.get("hardware", {})
        self.mode = hw_cfg.get("mode", "fake")
        self.host = hw_cfg.get("broker_host", "127.0.0.1")
        self.port = int(hw_cfg.get("broker_port", 1883))
        self.device_id = hw_cfg.get("device_id", "esp32-01")

        self.status_topic = f"aegis/status/{self.device_id}"
        self.event_topic = f"aegis/status/{self.device_id}/event"

        self.user = os.environ.get("AEGIS_MQTT_USER", "")
        self.password = os.environ.get("AEGIS_MQTT_PASS", "")

    def _write_device_status(self, is_online: bool | None = None):
        status_file = self.data_dir / "device_status.json"
        now_iso = datetime.now(timezone.utc).isoformat()
        current = {}
        if status_file.exists():
            try:
                current = json.loads(status_file.read_text(encoding="utf-8"))
            except Exception:
                current = {}

        current["broker_connected"] = bool(self.broker_connected)
        if is_online is not None:
            if is_online != self._last_state:
                self._last_state = is_online
                self._since = now_iso
            current["online"] = is_online
            current["device_online"] = is_online
            current["last_seen"] = now_iso
            current["since"] = self._since or now_iso
            current["timestamp"] = time.time()
            if self.mode in ("auto", "serial"):
                current["transport"] = "usb_serial"
                current["port"] = "COM11"
                current["description"] = "USB Serial ESP32 Device"

        tmp_file = status_file.with_name(f"device_status.tmp.{secrets.token_hex(4)}")
        try:
            with open(tmp_file, "w", encoding="utf-8") as f:
                f.write(json.dumps(current, indent=2))
                f.flush()
                try:
                    os.fsync(f.fileno())
                except Exception:
                    pass
            os.replace(tmp_file, status_file)
        except Exception:
            pass

    def start(self):
        if self.mode in ("auto", "serial"):
            try:
                bridge = security.hardware.get_bridge()
                if isinstance(bridge, security.hardware.SerialHardwareBridge):
                    is_on = bridge.is_device_online()
                    self.broker_connected = True
                    self._write_device_status(is_online=is_on)
                    self.is_running = True
                    print(f"[WORKER] USB Serial hardware subscriber active (port={getattr(bridge, '_detected_port', 'COM11')}, online={is_on})")
                else:
                    self.broker_connected = True
                    self._write_device_status(is_online=bridge.is_device_online())
                    self.is_running = True
            except Exception as e:
                print(f"[WORKER] USB Serial initialization note: {e}")
            return

        if self.mode != "mqtt":
            return

        try:
            import paho.mqtt.client as mqtt

            client_id = f"aegis_worker_{secrets.token_hex(4)}"
            if hasattr(mqtt, "CallbackAPIVersion"):
                self.client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2, client_id=client_id)
            else:
                self.client = mqtt.Client(client_id=client_id)

            if self.user and self.password:
                self.client.username_pw_set(self.user, self.password)

            self.client.on_connect = self._on_connect
            self.client.on_disconnect = self._on_disconnect
            self.client.on_message = self._on_message

            self.client.connect(self.host, self.port, keepalive=60)
            self.client.loop_start()
            self.is_running = True
            print(f"[WORKER] MQTT subscriber active on {self.status_topic}")
        except Exception as e:
            self.broker_connected = False
            self._write_device_status()
            print(f"[WORKER] MQTT subscriber could not connect ({e}). Continuing.")

    def _on_connect(self, client, userdata, flags, reason_code, properties=None):
        if reason_code == 0:
            self.broker_connected = True
            self._write_device_status()
            client.subscribe(self.status_topic, qos=1)
            client.subscribe(self.event_topic, qos=1)
        else:
            self.broker_connected = False
            self._write_device_status()

    def _on_disconnect(self, client, userdata, flags_or_rc=None, reason_code=None, properties=None):
        self.broker_connected = False
        self._write_device_status()

    def _on_message(self, client, userdata, msg):
        try:
            payload = msg.payload.decode("utf-8").strip()
            now_iso = datetime.now(timezone.utc).isoformat()

            if msg.topic == self.status_topic:
                is_online = (payload.lower() == "online")
                self._write_device_status(is_online=is_online)

            elif msg.topic == self.event_topic:
                events_file = self.data_dir / "device_events.jsonl"
                try:
                    event_data = json.loads(payload)
                except Exception:
                    event_data = {"event": payload}

                if "timestamp" not in event_data:
                    event_data["timestamp"] = now_iso

                # Read existing lines, cap at last 500
                existing_lines = []
                if events_file.exists():
                    try:
                        existing_lines = events_file.read_text(encoding="utf-8").strip().splitlines()
                    except Exception:
                        existing_lines = []

                existing_lines.append(json.dumps(event_data))
                capped = existing_lines[-500:]

                tmp_events = events_file.with_name(f"device_events.tmp.{secrets.token_hex(4)}")
                with open(tmp_events, "w", encoding="utf-8") as f:
                    f.write("\n".join(capped) + "\n")
                os.replace(tmp_events, events_file)

        except Exception as e:
            print(f"[WORKER] MQTT message error: {e}")

    def stop(self):
        self.broker_connected = False
        self._write_device_status()
        if self.client and self.is_running:
            try:
                self.client.loop_stop()
                self.client.disconnect()
            except Exception:
                pass
            self.is_running = False


class WorkerHeartbeatThread(threading.Thread):
    """Daemon thread writing heartbeat every 2s and refreshing ollama_ok every 15s."""

    def __init__(self, data_dir: Path, started_at_iso: str):
        super().__init__(daemon=True)
        self.data_dir = data_dir
        self.started_at_iso = started_at_iso
        self.running = True
        self.busy = False
        self.current_request_id = None
        self.ollama_ok = False
        self.model_warm = False
        self._last_ollama_check = 0.0
        self._last_hw_check = 0.0

    def run(self):
        while self.running:
            now = time.time()
            if now - self._last_ollama_check >= 15.0:
                self._refresh_ollama()
                self._last_ollama_check = now

            if now - self._last_hw_check >= 3.0:
                self._refresh_hardware()
                self._last_hw_check = now

            self.write_heartbeat()
            time.sleep(2.0)

    def _refresh_hardware(self):
        try:
            hw_cfg = security.policy_loader.POLICY.get("hardware", {})
            mode = hw_cfg.get("mode", "fake")
            if mode in ("auto", "serial"):
                bridge = security.hardware.get_bridge()
                if hasattr(bridge, "is_device_online"):
                    bridge.is_device_online()
        except Exception:
            pass

    def _refresh_ollama(self):
        try:
            client = ollama.Client(timeout=2.0)
            client.list()
            self.ollama_ok = True
        except Exception:
            self.ollama_ok = False

    def write_heartbeat(self):
        status_data = {
            "busy": self.busy,
            "current_request_id": self.current_request_id,
            "last_heartbeat": datetime.now(timezone.utc).isoformat(),
            "model_warm": self.model_warm,
            "ollama_ok": self.ollama_ok,
            "pid": os.getpid(),
            "started_at": self.started_at_iso,
        }
        status_file = self.data_dir / "worker_status.json"
        tmp_file = status_file.with_name(f"worker_status.tmp.{secrets.token_hex(4)}")
        try:
            with open(tmp_file, "w", encoding="utf-8") as f:
                f.write(json.dumps(status_data, indent=2))
                f.flush()
                try:
                    os.fsync(f.fileno())
                except Exception:
                    pass
            os.replace(tmp_file, status_file)
        except Exception:
            pass

    def set_busy(self, busy: bool, request_id: str | None = None):
        self.busy = busy
        self.current_request_id = request_id
        self.write_heartbeat()

    def stop(self):
        self.running = False

def _process_direct_tool(req: dict, emit, outbox_file: Path):
    raw_tool = req.get("tool")
    if isinstance(raw_tool, str):
        tool_name = raw_tool
        tool_args = {}
    elif isinstance(raw_tool, dict):
        tool_name = raw_tool.get("name", "")
        tool_args = raw_tool.get("args", {}) or {}
    else:
        tool_name = ""
        tool_args = {}

    user_text = req.get("text") or f"Direct execution of {tool_name}"
    role = req.get("role", "guest")
    session_id = req.get("session_id", "direct_tool_session")

    emit("detection", "skip", {"reason": "direct_tool bypasses prompt injection scan"})
    emit("llm", "skip", {"reason": "direct_tool bypasses LLM"})

    # Evaluate permissions, rate limits, and approvals via gateway.evaluate without duplicate checks
    eval_res = security.gateway.evaluate(
        tool_name=tool_name,
        tool_args=tool_args,
        user_text=user_text,
        role=role,
        session_id=session_id,
        emit=emit,
    )
    decision = eval_res.get("decision")
    reason = eval_res.get("reason", "")

    if decision != "ALLOW":
        for s in ("hardware", "output_scan"):
            emit(s, "skip", {"reason": f"Blocked: {reason}"})
        final = {
            "decision": "BLOCK",
            "error": reason,
            "final": True,
            "reply": f"Request blocked by policy: {reason}",
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }
        with open(outbox_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(final) + "\n")
            f.flush()
        return

    # Hardware execution
    emit("hardware", "running", {"args": tool_args, "tool": tool_name})
    try:
        if tool_name not in AVAILABLE_FUNCTIONS:
            raise RuntimeError(f"Tool '{tool_name}' has no registered execution function")
        expected_args = TOOL_SCHEMAS.get(tool_name, set())
        clean_args = {k: v for k, v in tool_args.items() if k in expected_args}
        tool_res = AVAILABLE_FUNCTIONS[tool_name](**clean_args)
        emit("hardware", "pass", {"result": str(tool_res), "tool": tool_name})
    except Exception as e:
        emit("hardware", "error", {"error": str(e), "tool": tool_name})
        emit("output_scan", "skip", {"reason": "Hardware execution error"})
        final = {
            "decision": "ERROR",
            "error": str(e),
            "final": True,
            "reply": f"Execution error: {e}",
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }
        with open(outbox_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(final) + "\n")
            f.flush()
        return

    # Output scan
    emit("output_scan", "pass", {})

    final = {
        "decision": "ALLOW",
        "error": None,
        "final": True,
        "reply": f"Executed {tool_name} successfully: {tool_res}",
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "tool_called": tool_name,
        "tool_result": tool_res,
    }
    with open(outbox_file, "a", encoding="utf-8") as f:
        f.write(json.dumps(final) + "\n")

def run_worker():

    outbox_dir = data_dir / "chat" / "outbox"
    inbox_dir.mkdir(parents=True, exist_ok=True)
    outbox_dir.mkdir(parents=True, exist_ok=True)

    lock_file_path = data_dir / "worker.lock"
    lock_fd = None
    try:
        lock_fd = open(lock_file_path, "a+")
        if os.name == "nt":
            import msvcrt
            msvcrt.locking(lock_fd.fileno(), msvcrt.LK_NBLCK, 1)
        else:
            import fcntl
            fcntl.flock(lock_fd.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        lock_fd.seek(0)
        lock_fd.truncate()
        lock_fd.write(f"{os.getpid()}\n")
        lock_fd.flush()
    except (OSError, IOError) as e:
        print(f"[WORKER] Fatal: Another worker is already running (lock held at {lock_file_path}: {e}). Exiting.")
        sys.exit(1)

    started_at_iso = datetime.now(timezone.utc).isoformat()
    print("=" * 60)
    print(" AegisAgent Worker Service ".center(60, "="))
    print(f" PID: {os.getpid()} | Started: {started_at_iso}")
    print(f" Polling: {inbox_dir}")
    print("=" * 60)

    ollama_ok = False
    model_warm = False
    print("[WORKER] Warming up Ollama model...")
    try:
        ollama.chat(model=MODEL, messages=[{"role": "user", "content": "hi"}], keep_alive="30m")
        ollama_ok = True
        model_warm = True
        print("[WORKER] Model warmed up and ready.")
    except Exception as e:
        print(f"[WORKER] Notice: Ollama not reachable on start ({e}). Continuing.")

    # Start Heartbeat Daemon Thread
    hb_thread = WorkerHeartbeatThread(data_dir, started_at_iso)
    hb_thread.ollama_ok = ollama_ok
    hb_thread.model_warm = model_warm
    hb_thread.write_heartbeat()
    hb_thread.start()

    # Load seen nonces
    seen_nonces_list = _load_seen_nonces(data_dir)
    seen_nonces_set = set(seen_nonces_list)

    # Start MQTT Subscriber
    device_subscriber = DeviceStatusSubscriber(data_dir)
    device_subscriber.start()

    running = True

    def _sig_handler(sig, frame):
        nonlocal running
        print("\n[WORKER] Shutdown signal received. Exiting...")
        running = False

    signal.signal(signal.SIGINT, _sig_handler)
    if hasattr(signal, "SIGTERM"):
        signal.signal(signal.SIGTERM, _sig_handler)

    while running:
        inbox_files = sorted(inbox_dir.glob("*.json"))
        for req_file in inbox_files:
            stem = req_file.stem
            is_stem_valid = bool(ID_PATTERN.match(stem))

            try:
                content = req_file.read_text(encoding="utf-8").strip()
                if not content:
                    try:
                        req_file.unlink()
                    except Exception:
                        pass
                    continue
                req = json.loads(content)
            except Exception as e:

                    "worker_rejected",

                        "error": str(e),
                        "file": req_file.name,

                try:
                    req_file.unlink()
                except Exception:
                    pass
                continue

            # Validate ID against ID_PATTERN
            req_id = req.get("id")
            is_id_valid = isinstance(req_id, str) and bool(ID_PATTERN.match(req_id))

            # Validate kind
            kind = req.get("kind", "chat")
            is_kind_valid = kind in ("chat", "direct_tool")

            # Validate role defined in policy.yaml
            valid_roles = set(security.policy_loader.POLICY.get("roles", {}).keys())
            role = req.get("role", "guest")
            is_role_valid = isinstance(role, str) and (role in valid_roles)

            # Validate timestamp (epoch seconds, |now - ts| <= 60s)
            ts = req.get("ts")
            is_ts_valid = False
            if ts is not None and isinstance(ts, (int, float)):
                if abs(time.time() - float(ts)) <= 60.0:
                    is_ts_valid = True

            # Validate nonce (16 random bytes hex = 32 hex chars)
            nonce = req.get("nonce")
            is_nonce_valid = (
                isinstance(nonce, str)
                and len(nonce) == 32
                and all(c in "0123456789abcdefABCDEF" for c in nonce)
            )
            is_nonce_new = is_nonce_valid and (nonce not in seen_nonces_set)

            # Check signature
            sig = req.get("signature")
            sign_payload = {k: v for k, v in req.items() if k != "signature"}
            is_sig_valid = bool(sig and security.signing.verify(sign_payload, sig))

            # Determine rejection reason
            reject_reason = None
            reject_detail = None

            if not is_sig_valid:
                reject_reason = "bad_or_missing_signature"
                reject_detail = "invalid or missing cryptographic signature"
            elif not is_id_valid:
                reject_reason = "invalid_request_id"
                reject_detail = "request id format is invalid"
            elif not is_kind_valid:
                reject_reason = "unknown_kind"
                reject_detail = f"unknown request kind '{kind}'"
            elif not is_role_valid:
                reject_reason = "invalid_role"
                reject_detail = f"role '{role}' is not defined in policy"
            elif not is_ts_valid:
                reject_reason = "stale_timestamp"
                reject_detail = "request timestamp is missing or outside 60-second window"
            elif not is_nonce_valid:
                reject_reason = "invalid_nonce"
                reject_detail = "request nonce is missing or not a 16-byte hex token"
            elif not is_nonce_new:
                reject_reason = "replayed_nonce"
                reject_detail = "request nonce has already been seen (replay detected)"

            if reject_reason is not None:
                security.logger.log_event(
                    "worker_rejected",
                    {
                        "file": req_file.name,
                        "id": req_id if is_id_valid else "invalid",

                        "username": req.get("username", "unknown"),
                    },
                )

                # If stem is invalid, write no outbox file at all.
                if is_stem_valid:
                    outbox_file = outbox_dir / f"{stem}.jsonl"
                    _write_rejection_outbox(outbox_file, reject_detail)

                try:
                    req_file.unlink()
                except Exception:
                    pass
                continue

            # Record seen nonce and persist atomically
            seen_nonces_set.add(nonce)
            seen_nonces_list.append(nonce)
            if len(seen_nonces_list) > 2000:
                seen_nonces_list = seen_nonces_list[-2000:]
                seen_nonces_set = set(seen_nonces_list)
            _save_seen_nonces(data_dir, seen_nonces_list)

            # Valid request: build outbox path using validated req_id
            outbox_file = outbox_dir / f"{req_id}.jsonl"

            # Remove inbox file once read
            try:
                req_file.unlink()
            except Exception:
                pass

            def emit(stage: str, status: str, detail: dict):
                step = {
                    "detail": detail,
                    "stage": stage,
                    "status": status,
                    "timestamp": datetime.now(timezone.utc).isoformat(),
                }
                with open(outbox_file, "a", encoding="utf-8") as f:
                    f.write(json.dumps(step) + "\n")
                    f.flush()

            # Process request with busy state in heartbeat thread
            hb_thread.set_busy(True, req_id)
            try:
                if kind == "direct_tool":
                    _process_direct_tool(req, emit, outbox_file)
                else:
                    assigned_role = role
                    user_text = req.get("text", "")
                    session_id = req.get("session_id", "default")

                    res = run_turn(text=user_text, role=assigned_role, session_id=session_id, emit=emit)
                    final = {
                        "decision": res.decision,
                        "error": res.error,
                        "final": True,
                        "reply": res.reply,
                        "timestamp": datetime.now(timezone.utc).isoformat(),
                        "tool_called": res.tool_called,
                        "tool_result": res.tool_result,
                    }
                    with open(outbox_file, "a", encoding="utf-8") as f:
                        f.write(json.dumps(final) + "\n")
                        f.flush()
            except Exception as e:
                print(f"[WORKER] Error processing request {req_id}: {e}")
                err_final = {
                    "decision": "ERROR",
                    "error": str(e),
                    "final": True,
                    "reply": f"Worker encountered an internal error: {e}",
                    "timestamp": datetime.now(timezone.utc).isoformat(),
                }
                with open(outbox_file, "a", encoding="utf-8") as f:
                    f.write(json.dumps(err_final) + "\n")
                    f.flush()
            finally:
                hb_thread.set_busy(False, None)

        time.sleep(0.3)

    hb_thread.stop()
    device_subscriber.stop()
    if lock_fd:
        try:
            if os.name == "nt":
                import msvcrt
                lock_fd.seek(0)
                msvcrt.locking(lock_fd.fileno(), msvcrt.LK_UNLCK, 1)
            else:
                import fcntl
                fcntl.flock(lock_fd.fileno(), fcntl.LOCK_UN)
            lock_fd.close()
        except Exception:
            pass
    print("[WORKER] Worker stopped gracefully.")


if __name__ == "__main__":
    run_worker()

                hb_dt = datetime.fromisoformat(last_hb_str)
                if hb_dt.tzinfo is None:
                    hb_dt = hb_dt.replace(tzinfo=timezone.utc)
                age = (now_utc - hb_dt).total_seconds()
                # Worker considered alive if heartbeat is less than 10s old
                if age < 10.0:
                    worker_alive = True

            current_request_id = w_data.get("current_request_id")
            worker_pid = w_data.get("pid")
            model_warm = bool(w_data.get("model_warm", False))
            ollama_ok = bool(w_data.get("ollama_ok", False))

            if worker_alive:
                state_str = "Busy" if worker_busy else "Idle"
                worker_detail = f"PID {worker_pid} ({state_str})"

    device_file = data_dir / "device_status.json"

            broker_connected = False
    else:

        device_status = "unknown"
        broker_connected = False

    # 3. Ollama Status
    ollama_status = "ready" if ollama_ok else "offline"

    # 4. Broker Status
    broker_status = "connected" if broker_connected else "disconnected"

    # 5. Audit Chain Verification (cached, re-verified at most once per 30s)
    if force_audit_check or _audit_cache is None or (now - _audit_cache[2]) >= 30.0:

"""
Phase 7 ƒ?" Hardware Bridge & MQTT Integration Framework
"""

from dataclasses import dataclass
from datetime import datetime, timezone
import json

@dataclass
class HardwareCommandResult:
    ok: bool
    error: str | None = None


class HardwareBridge:
    def __init__(self, policy_dict: dict | None = None):
        if policy_dict is None:
            policy_dict = policy_loader.POLICY

        hw_cfg = policy_dict.get("hardware", {})
        self.host = hw_cfg.get("broker_host", "127.0.0.1")
        self.port = int(hw_cfg.get("broker_port", 1883))
        self.device_id = hw_cfg.get("device_id", "esp32-01")
        self.timeout = float(hw_cfg.get("publish_timeout_seconds", 5))

        self.user = os.environ.get("AEGIS_MQTT_USER", "")
        self.password = os.environ.get("AEGIS_MQTT_PASS", "")

        self.cmd_topic = f"aegis/cmd/{self.device_id}"
        self.status_topic = f"aegis/status/{self.device_id}"

        self.device_online = False
        self.client = None
        self._connect_event = threading.Event()
        self._status_event = threading.Event()
        self._connected = False
        self._connect_error = None

    def _create_client(self) -> mqtt.Client:
        client_id = f"aegis_gateway_{uuid.uuid4().hex[:8]}"
        if hasattr(mqtt, "CallbackAPIVersion"):
            client = mqtt.Client(
                mqtt.CallbackAPIVersion.VERSION2,
                client_id=client_id,
            )
        else:
            client = mqtt.Client(
                client_id=client_id,
            )

        if self.user and self.password:
            client.username_pw_set(self.user, self.password)

        client.on_connect = self._on_connect
        client.on_message = self._on_message
        if hasattr(client, "on_connect_fail"):
            client.on_connect_fail = self._on_connect_fail

        return client

    def _on_connect(self, client, userdata, flags, reason_code, properties=None):
        if reason_code == 0:
            self._connected = True
            client.subscribe(self.status_topic, qos=1)
            self._connect_event.set()
        else:
            self._connected = False
            self._connect_error = f"MQTT connection rejected (reason code: {reason_code})"
            self._connect_event.set()

    def _on_connect_fail(self, client, userdata):
        self._connected = False
        self._connect_error = "MQTT broker connection failed"
        self._connect_event.set()

    def _save_device_status(self):
        try:
            status_dir = _get_data_dir()
            status_dir.mkdir(parents=True, exist_ok=True)
            status_file = status_dir / "device_status.json"
            tmp_file = status_dir / f"device_status.tmp.{uuid.uuid4().hex[:6]}"
            data = {
                "device_online": self.device_online,
                "device_id": self.device_id,
                "timestamp": time.time(),
            }
            with open(tmp_file, "w", encoding="utf-8") as f:
                f.write(json.dumps(data, indent=2))
            os.replace(tmp_file, status_file)
        except Exception:
            pass

    def _on_message(self, client, userdata, msg):
        try:
            if msg.topic == self.status_topic:
                payload = msg.payload.decode("utf-8").strip()
                self.device_online = (payload == "online")
                self._save_device_status()
                self._status_event.set()
        except Exception:
            pass

    def is_device_online(self) -> bool:
        return self.device_online

    def publish(self, command: str) -> HardwareCommandResult:
        if command not in COMMAND_ALLOWLIST:
            raise ValueError(f"Command '{command}' is not in hardware command allowlist")

        ok = False
        error_msg = None

        self._connect_event.clear()
        self._status_event.clear()
        self._connected = False
        self._connect_error = None
        self.device_online = False

        client = self._create_client()
        self.client = client

        try:
            connected_socket = False
            try:
                client.connect(self.host, self.port, keepalive=60)
                client.loop_start()
                connected_socket = True
            except Exception as e:
                ok = False
                error_msg = f"MQTT connection error: {e}"

            if connected_socket:
                if not self._connect_event.wait(timeout=5.0):
                    ok = False
                    error_msg = "MQTT connection timed out waiting for CONNACK"
                elif not self._connected:
                    ok = False
                    error_msg = self._connect_error or "MQTT connection failed"
                else:
                    # Subscribe to status topic and wait up to about 3 s for retained message
                    client.subscribe(self.status_topic, qos=1)
                    self._status_event.wait(timeout=3.0)
                    self._save_device_status()

                    if not self.device_online:
                        ok = False
                        error_msg = "Device offline - command not delivered"
                    else:
                        payload_dict = {
                            "command": command,
                            "ts": time.time(),
                            "nonce": str(uuid.uuid4()),
                        }
                        payload_str = json.dumps(payload_dict)

                        try:
                            info = client.publish(self.cmd_topic, payload_str, qos=1)
                            info.wait_for_publish(timeout=self.timeout)
                            if info.is_published():
                                ok = True
                            else:
                                ok = False
                                error_msg = "MQTT publish acknowledgement timed out"
                        except Exception as e:
                            ok = False
                            error_msg = f"MQTT publish failed: {e}"

        except Exception as e:
            ok = False
            error_msg = f"MQTT error: {e}"
        finally:
            try:
                client.loop_stop()
            except Exception:
                pass
            try:
                client.disconnect()
            except Exception:
                pass

        logger.log_hardware_command(command, ok, error_msg)
        return HardwareCommandResult(ok=ok, error=error_msg)

class SerialHardwareBridge:
    """
    Direct USB Serial transport for ESP32 hardware device.
    Enables plug-and-play local hardware communication without external broker or Wi-Fi dependencies.
    """
    def __init__(self, port: str | None = None, baud: int = 115200, policy_dict: dict | None = None):
        if policy_dict is None:
            policy_dict = policy_loader.POLICY

        hw_cfg = policy_dict.get("hardware", {})
        self.port = port or hw_cfg.get("serial_port", "auto")
        self.baud = int(hw_cfg.get("baud_rate", baud))
        self.device_id = hw_cfg.get("device_id", "esp32-01")
        self.device_online = False
        self._lock = threading.RLock()
        self._serial = None
        self._detected_port = None
        self._detected_desc = None
        self._last_opened_at = 0.0
        self._try_connect()

    def _detect_port(self) -> str | None:
        try:
            import serial.tools.list_ports
            ports = list(serial.tools.list_ports.comports())
            for p in ports:
                desc = (p.description or "").lower()
                hwid = (p.hwid or "").lower()
                if any(k in desc or k in hwid for k in ["cp210", "ch340", "ftdi", "usb to uart", "esp32", "silicon labs"]):
                    self._detected_desc = p.description
                    return p.device
            for p in ports:
                if "usb" in (p.hwid or "").lower() or "usb" in (p.description or "").lower():
                    self._detected_desc = p.description
                    return p.device
        except Exception:
            pass
        return None

    def _format_port_name(self, port_str: str) -> str:
        if os.name == "nt" and port_str.upper().startswith("COM"):
            try:
                num = int(port_str[3:])
                if num >= 10 and not port_str.startswith(r"\\.\ "):
                    return rf"\\.\{port_str}"
            except ValueError:
                pass
        return port_str

    def _try_connect(self) -> bool:
        with self._lock:
            if self._serial is not None and getattr(self._serial, "is_open", False):
                self.device_online = True
                return True
            target_port = self.port
            if not target_port or target_port == "auto":
                target_port = self._detect_port()
            if not target_port:
                self.device_online = False
                self._save_device_status()
                return False

            self._detected_port = target_port
            fmt_port = self._format_port_name(target_port)
            try:
                import serial
                s = serial.Serial()
                s.port = fmt_port
                s.baudrate = self.baud
                s.timeout = 0.25
                s.write_timeout = 1.0
                try:
                    s.dtr = False
                    s.rts = False
                except Exception:
                    pass
                s.open()
                self._serial = s
                self._last_opened_at = time.time()
                self.device_online = True
                self._save_device_status()
                return True
            except Exception:
                self.device_online = False
                self._save_device_status()
                return False

    def _save_device_status(self):
        try:
            status_dir = _get_data_dir()
            status_dir.mkdir(parents=True, exist_ok=True)
            status_file = status_dir / "device_status.json"
            tmp_file = status_dir / f"device_status.tmp.{uuid.uuid4().hex[:6]}"
            now_iso = datetime.now(timezone.utc).isoformat()
            data = {
                "online": self.device_online,
                "device_online": self.device_online,
                "broker_connected": True,
                "transport": "usb_serial",
                "port": self._detected_port or self.port or "COM11",
                "description": self._detected_desc or "USB Serial ESP32 Device",
                "device_id": self.device_id,
                "timestamp": time.time(),
                "last_seen": now_iso,
                "since": now_iso,
            }
            with open(tmp_file, "w", encoding="utf-8") as f:
                f.write(json.dumps(data, indent=2))
            os.replace(tmp_file, status_file)
        except Exception:
            pass

    def is_device_online(self) -> bool:
        with self._lock:
            if self._serial is not None and getattr(self._serial, "is_open", False):
                return True
            return self._try_connect()

    def publish(self, command: str) -> HardwareCommandResult:
        if command not in COMMAND_ALLOWLIST:
            raise ValueError(f"Command '{command}' is not in hardware command allowlist")

        with self._lock:
            if not self._serial or not getattr(self._serial, "is_open", False):
                if not self._try_connect():
                    logger.log_hardware_command(command, False, "USB hardware not connected")
                    return HardwareCommandResult(ok=False, error="USB hardware not connected")

            try:
                # If newly opened, wait a moment for ESP32 boot banner if needed
                if time.time() - self._last_opened_at < 1.4:
                    time.sleep(max(0.05, 1.4 - (time.time() - self._last_opened_at)))

                try:
                    self._serial.reset_input_buffer()
                except Exception:
                    pass

                line_to_send = f"{command}\n".encode("utf-8")
                self._serial.write(line_to_send)

                t0 = time.time()
                while time.time() - t0 < 0.6:
                    if self._serial.in_waiting:
                        chunk = self._serial.read(self._serial.in_waiting).decode("utf-8", errors="ignore")
                        if any(k in chunk for k in ("EVENT:", "USB_RECV:", "done", "door", "alarm", "alert", "STATUS:")):
                            break
                    time.sleep(0.02)

                self.device_online = True
                self._save_device_status()
                logger.log_hardware_command(command, True, None)
                return HardwareCommandResult(ok=True)
            except Exception as e:
                self.device_online = False
                try:
                    if self._serial:
                        self._serial.close()
                except Exception:
                    pass
                self._serial = None
                self._save_device_status()
                logger.log_hardware_command(command, False, str(e))
                return HardwareCommandResult(ok=False, error=str(e))

"""
Phase 4 & Phase 8A ƒ?" Tamper-Evident Audit Logger

WHAT THIS FILE DOES:
Appends decision and security logs to logs/audit_log.jsonl using a SHA-256 hash chain.
Each log line includes a cryptographic hash of its contents plus the previous line's hash.
Supports cross-process file locking (Windows msvcrt, POSIX fcntl), configurable log path,
and log rotation/archival.
"""

from contextlib import contextmanager
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import shutil
import time

_custom_log_file: Path | None = None


def get_log_file() -> Path:

    env_path = os.environ.get("AEGIS_LOG_FILE")

def __getattr__(name):

        return get_log_file()
    if name == "LOG_DIR":
        return get_log_file().parent
    raise AttributeError(f"module {__name__!r} has no attribute {name!r}")


@contextmanager
def _file_lock(log_path: Path, timeout: float = 10.0):
    """
    Cross-process file lock using msvcrt on Windows and fcntl on POSIX.
    Locks a companion .lock file to avoid conflicting with readers of the log file.
    """
    lock_path = log_path.with_name(log_path.name + ".lock")
    lock_path.parent.mkdir(parents=True, exist_ok=True)
    
    start_time = time.time()
    f = open(lock_path, "a+b")
    try:
        if os.name == "nt":
            import msvcrt
            while True:
                try:
                    f.seek(0)
                    msvcrt.locking(f.fileno(), msvcrt.LK_NBLCK, 1)
                    break
                except (OSError, PermissionError, IOError):
                    if time.time() - start_time > timeout:
                        raise TimeoutError(f"Could not acquire file lock on {lock_path} within {timeout}s")
                    time.sleep(0.01)
            try:
                yield
            finally:
                try:
                    f.seek(0)
                    msvcrt.locking(f.fileno(), msvcrt.LK_UNLCK, 1)
                except Exception:
                    pass
        else:
            import fcntl
            while True:
                try:
                    fcntl.flock(f.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
                    break
                except (OSError, IOError, BlockingIOError):
                    if time.time() - start_time > timeout:
                        raise TimeoutError(f"Could not acquire file lock on {lock_path} within {timeout}s")
                    time.sleep(0.01)
            try:
                yield
            finally:
                try:
                    fcntl.flock(f.fileno(), fcntl.LOCK_UN)
                except Exception:
                    pass
    finally:
        try:
            f.close()
        except Exception:
            pass


def _get_last_entry_hash(log_file: Path | None = None) -> str:
    """Returns the entry_hash of the last line in the log file, or '0'*64 if empty/missing."""
    if log_file is None:
        log_file = get_log_file()

    if not log_file.exists():
        return "0" * 64

    try:
        content = log_file.read_text(encoding="utf-8").strip()
        if not content:
            return "0" * 64
        lines = content.splitlines()
        if not lines:
            return "0" * 64
        last_entry = json.loads(lines[-1])
        return last_entry.get("entry_hash", "0" * 64)
    except Exception:
        return "0" * 64


def _append_entry(entry_data: dict) -> dict:
    """
    Appends an entry to the hash chain under a cross-process lock.
    Safely executes: read last hash -> append line -> flush (+ fsync).
    """
    log_file = get_log_file()
    log_file.parent.mkdir(parents=True, exist_ok=True)

    with _file_lock(log_file):
        prev_hash = _get_last_entry_hash(log_file)
        
        # Build entry without hash fields
        entry_without_hash = {k: v for k, v in entry_data.items() if k not in ("prev_hash", "entry_hash")}
        serialized_data = json.dumps(entry_without_hash, sort_keys=True)
        payload = serialized_data + prev_hash
        entry_hash = hashlib.sha256(payload.encode("utf-8")).hexdigest()

        log_entry = dict(entry_without_hash)
        log_entry["prev_hash"] = prev_hash
        log_entry["entry_hash"] = entry_hash

        with open(log_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(log_entry) + "\n")
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass

        return log_entry


def archive_and_reset(reason: str) -> dict:
    """
    Moves current log to logs/archive/<UTC timestamp>.jsonl and starts a new
    chain whose first entry records the archived file name and its final hash.
    """
    log_file = get_log_file()
    log_dir = log_file.parent
    archive_dir = log_dir / "archive"
    archive_dir.mkdir(parents=True, exist_ok=True)

    with _file_lock(log_file):
        final_hash = _get_last_entry_hash(log_file)
        now_utc = datetime.now(timezone.utc)
        ts_str = now_utc.strftime("%Y-%m-%dT%H-%M-%SZ")
        archive_name = f"{ts_str}.jsonl"
        archive_path = archive_dir / archive_name
        if archive_path.exists():
            archive_name = f"{ts_str}_{now_utc.microsecond:06d}.jsonl"
            archive_path = archive_dir / archive_name

        if log_file.exists():
            shutil.move(str(log_file), str(archive_path))
        else:
            archive_name = "none"

        # Start a new chain
        first_entry_data = {
            "archived_file": archive_name,
            "event": "log_archive_reset",
            "final_hash": final_hash,
            "reason": reason,
            "timestamp": now_utc.isoformat(),
        }

        prev_hash = "0" * 64
        serialized_data = json.dumps(first_entry_data, sort_keys=True)
        payload = serialized_data + prev_hash
        entry_hash = hashlib.sha256(payload.encode("utf-8")).hexdigest()

        first_entry = dict(first_entry_data)
        first_entry["prev_hash"] = prev_hash
        first_entry["entry_hash"] = entry_hash

        with open(log_file, "w", encoding="utf-8") as f:
            f.write(json.dumps(first_entry) + "\n")
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass

        return first_entry

"""
Phase 8A ƒ?" Authentication & User Management Framework

WHAT THIS FILE DOES:
Manages user accounts, PBKDF2 password hashing (200,000 iterations), constant-time
login verification, per-user lockout policy (5 failed attempts -> 5-minute lockout),
and audit logging for all authentication events (login_success, login_failure, logout, user_created).
"""

from datetime import datetime, timezone
import hashlib
import hmac
import json
import os
from pathlib import Path
import secrets
import time

import security.logger
import security.policy_loader

LOCKOUT_MAX_ATTEMPTS = 5
LOCKOUT_DURATION_SECONDS = 300.0  # 5 minutes


def _get_users_file() -> Path:
    data_dir = Path(os.environ.get("AEGIS_DATA_DIR", Path(__file__).resolve().parent.parent / "data"))
    data_dir.mkdir(parents=True, exist_ok=True)
    return data_dir / "users.json"


def _load_data() -> dict:
    users_file = _get_users_file()
    if not users_file.exists():
        return {"users": {}, "lockouts": {}}
    try:
        content = users_file.read_text(encoding="utf-8").strip()
        if not content:
            return {"users": {}, "lockouts": {}}
        data = json.loads(content)
        if "users" not in data:
            data["users"] = {}
        if "lockouts" not in data:
            data["lockouts"] = {}
        return data
    except Exception:
        return {"users": {}, "lockouts": {}}


def _save_data(data: dict):
    users_file = _get_users_file()
    tmp_file = users_file.with_name(f"users.tmp.{secrets.token_hex(4)}")
    with open(tmp_file, "w", encoding="utf-8") as f:
        f.write(json.dumps(data, indent=2))
        f.flush()
        try:
            os.fsync(f.fileno())
        except Exception:
            pass
    os.replace(tmp_file, users_file)


def _hash_password(password: str, salt_hex: str) -> str:
    salt_bytes = bytes.fromhex(salt_hex)
    return hashlib.pbkdf2_hmac(
        "sha256",
        password.encode("utf-8"),
        salt_bytes,
        200000,
    ).hex()


def create_user(username: str, password: str, role: str) -> dict:
    """
    Creates a new user account with a salted PBKDF2-SHA256 password hash.
    Role must exist in policy.yaml. Never logs the password.
    """
    username = username.strip()
    if not username:
        raise ValueError("Username cannot be empty")

    roles_cfg = security.policy_loader.POLICY.get("roles", {})
    if role not in roles_cfg:
        valid_roles = ", ".join(roles_cfg.keys())
        raise ValueError(f"Role '{role}' does not exist in policy.yaml. Valid roles: {valid_roles}")

    data = _load_data()
    if username in data["users"]:
        raise ValueError(f"User '{username}' already exists")

    salt_hex = secrets.token_hex(16)
    hash_hex = _hash_password(password, salt_hex)

    user_entry = {
        "created_at": datetime.now(timezone.utc).isoformat(),
        "pbkdf2_hash": hash_hex,
        "role": role,
        "salt": salt_hex,
        "username": username,
    }
    data["users"][username] = user_entry
    _save_data(data)

    security.logger.log_event("user_created", {"role": role, "username": username})

    return {
        "created_at": user_entry["created_at"],
        "role": user_entry["role"],
        "username": user_entry["username"],
    }


def reset_password(username: str, new_password: str) -> bool:
    """
    Resets the password and clears any lockout for an existing user.
    Audits password_reset event.
    """
    username = username.strip()
    data = _load_data()
    if username not in data.get("users", {}):
        raise ValueError(f"User '{username}' does not exist")
    salt_hex = secrets.token_hex(16)
    hash_hex = _hash_password(new_password, salt_hex)
    data["users"][username]["salt"] = salt_hex
    data["users"][username]["pbkdf2_hash"] = hash_hex
    data["users"][username]["updated_at"] = datetime.now(timezone.utc).isoformat()
    if "lockouts" in data and username in data["lockouts"]:
        del data["lockouts"][username]
    _save_data(data)
    security.logger.log_event("password_reset", {"username": username})
    return True


def verify_login(username: str, password: str) -> tuple[bool, str]:
    """
    Verifies user credentials using constant-time comparison.
    Enforces per-username account lockout after 5 consecutive failed attempts for 5 minutes.
    Returns (True, role) on success, or (False, error_message) on failure.
    Logs audit events: login_success, login_failure.
    """
    username = username.strip()
    data = _load_data()
    lockouts = data.get("lockouts", {})
    user_lockout = lockouts.get(username, {"failed_attempts": 0, "locked_until": 0.0})

    now = time.time()
    locked_until = float(user_lockout.get("locked_until", 0.0))

    # Check active lockout
    if locked_until > now:
        remaining_seconds = int(locked_until - now)
        security.logger.log_event(
            "login_failure",
            {"reason": "account_locked", "remaining_seconds": remaining_seconds, "username": username},
        )
        return False, f"Account is locked. Try again in {remaining_seconds} seconds."

    user = data.get("users", {}).get(username)

    if not user:
        # Constant-time dummy computation to defend against timing attacks
        dummy_salt = "0" * 32
        _hash_password(password, dummy_salt)
        security.logger.log_event(
            "login_failure",
            {"reason": "user_not_found", "username": username},
        )
        return False, "Invalid username or password"

    stored_hash = user.get("pbkdf2_hash", "")
    salt = user.get("salt", "")
    computed_hash = _hash_password(password, salt)

    if hmac.compare_digest(stored_hash, computed_hash):
        # Reset lockout counter on success
        if username in lockouts:
            del lockouts[username]
            _save_data(data)

        role = user.get("role", "guest")
        security.logger.log_event(
            "login_success",
            {"role": role, "username": username},
        )
        return True, role
    else:
        failed_attempts = int(user_lockout.get("failed_attempts", 0)) + 1
        if failed_attempts >= LOCKOUT_MAX_ATTEMPTS:
            new_locked_until = now + LOCKOUT_DURATION_SECONDS
            lockouts[username] = {
                "failed_attempts": failed_attempts,
                "locked_until": new_locked_until,
            }
            _save_data(data)
            security.logger.log_event(
                "login_failure",
                {
                    "failed_attempts": failed_attempts,
                    "reason": "lockout_triggered",
                    "username": username,
                },
            )
            return False, f"Account locked for {int(LOCKOUT_DURATION_SECONDS / 60)} minutes due to {LOCKOUT_MAX_ATTEMPTS} failed login attempts."
        else:
            lockouts[username] = {
                "failed_attempts": failed_attempts,
                "locked_until": 0.0,
            }
            _save_data(data)
            remaining_attempts = LOCKOUT_MAX_ATTEMPTS - failed_attempts
            security.logger.log_event(
                "login_failure",
                {
                    "failed_attempts": failed_attempts,
                    "reason": "bad_password",
                    "username": username,
                },
            )
            return False, f"Invalid username or password ({remaining_attempts} attempts remaining before lockout)."
